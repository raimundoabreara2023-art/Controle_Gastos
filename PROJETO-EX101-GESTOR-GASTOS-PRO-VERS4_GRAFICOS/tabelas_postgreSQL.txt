-- Table: public.categorias

-- DROP TABLE IF EXISTS public.categorias;

CREATE TABLE IF NOT EXISTS public.categorias
(
    id_categoria integer NOT NULL DEFAULT nextval('categorias_id_categoria_seq'::regclass),
    nome character varying(50) COLLATE pg_catalog."default" NOT NULL,
    CONSTRAINT categorias_pkey PRIMARY KEY (id_categoria),
    CONSTRAINT categorias_nome_key UNIQUE (nome)
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS public.categorias
    OWNER to postgres;

-- Table: public.subcategorias

-- DROP TABLE IF EXISTS public.subcategorias;

CREATE TABLE IF NOT EXISTS public.subcategorias
(
    id_subcategoria integer NOT NULL DEFAULT nextval('subcategorias_id_subcategoria_seq'::regclass),
    id_categoria integer NOT NULL,
    nome character varying(50) COLLATE pg_catalog."default" NOT NULL,
    CONSTRAINT subcategorias_pkey PRIMARY KEY (id_subcategoria),
    CONSTRAINT subcategorias_nome_categoria_unique UNIQUE (id_categoria, nome),
    CONSTRAINT subcategorias_id_categoria_fkey FOREIGN KEY (id_categoria)
        REFERENCES public.categorias (id_categoria) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE CASCADE
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS public.subcategorias
    OWNER to postgres;

-- Table: public.transacoes_detalhe

-- DROP TABLE IF EXISTS public.transacoes_detalhe;

CREATE TABLE IF NOT EXISTS public.transacoes_detalhe
(
    id_transacao integer NOT NULL DEFAULT nextval('transacoes_detalhe_id_transacao_seq'::regclass),
    id_compra integer,
    id_membro integer,
    id_mes_ref integer,
    valor_parcela numeric(10,2) NOT NULL,
    numero_parcela integer NOT NULL,
    descricao_formatada character varying(150) COLLATE pg_catalog."default",
    id_produto integer,
    quantidade numeric(10,3) NOT NULL DEFAULT 1,
    valor_unitario numeric(10,2) NOT NULL DEFAULT 0.00,
    CONSTRAINT transacoes_detalhe_pkey PRIMARY KEY (id_transacao),
    CONSTRAINT transacoes_detalhe_id_compra_fkey FOREIGN KEY (id_compra)
        REFERENCES public.compras_mestre (id_compra) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE CASCADE,
    CONSTRAINT transacoes_detalhe_id_membro_fkey FOREIGN KEY (id_membro)
        REFERENCES public.membros (id_membro) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION,
    CONSTRAINT transacoes_detalhe_id_mes_ref_fkey FOREIGN KEY (id_mes_ref)
        REFERENCES public.meses_referencia (id_mes) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION,
    CONSTRAINT transacoes_detalhe_id_produto_fkey FOREIGN KEY (id_produto)
        REFERENCES public.produtos (id_produto) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS public.transacoes_detalhe
    OWNER to postgres;
